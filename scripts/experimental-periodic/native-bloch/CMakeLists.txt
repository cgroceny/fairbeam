cmake_minimum_required(VERSION 3.16)
project(fairbeam_native_bloch_kernel LANGUAGES CXX)
enable_testing()
add_executable(native_bloch_tests test_bloch_kernel.cpp)
target_compile_features(native_bloch_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(native_bloch_tests PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(native_bloch_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME paired_bloch_kernel COMMAND native_bloch_tests)
